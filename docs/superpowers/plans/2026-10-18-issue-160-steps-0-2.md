```rust
use std::collections::BTreeMap;
use std::path::Path;
use std::process::{Command, Output};

use serde_json::Value;

/// Spawn `alsem <args>` with tracing on into a temp trace file, every other
/// `ALSEM_TRACE_*` knob removed (pattern: `cli_envelope_detector_set.rs:35-76`).
pub(crate) fn spawn_traced(args: &[&str], extra_env: &[(&str, &str)]) -> (Output, BTreeMap<String, u64>) {
    let dir = tempfile::tempdir().expect("tempdir");
    let trace = dir.path().join("trace.json");
    let mut cmd = Command::new(env!("CARGO_BIN_EXE_alsem"));
    cmd.args(args)
        .env_remove("ALSEM_TRACE_DETAIL")
        .env_remove("ALSEM_TRACE_SAMPLE_EVERY")
        .env_remove("ALSEM_TRACE_SCC_MIN")
        .env_remove("ALSEM_TRACE_STDERR")
        .env_remove("ALSEM_TRACE_EXIT_AFTER")
        .env_remove("ALSEM_CONES_CENSUS")
        .env_remove("ALSEM_RETAINED_CENSUS")
        .env("ALSEM_TRACE", "1")
        .env("ALSEM_TRACE_FILE", &trace);
    for (k, v) in extra_env {
        cmd.env(k, v);
    }
    let out = cmd.output().expect("spawn alsem");
    let text = std::fs::read_to_string(&trace).unwrap_or_default();
    let events: Vec<Value> = serde_json::from_str(&text).unwrap_or_default();
    (out, last_counters(&events))
}

/// LAST `C` event value per counter name (spec §4.1.2). Absent = 0 via `count()`.
fn last_counters(events: &[Value]) -> BTreeMap<String, u64> {
    let mut m = BTreeMap::new();
    for ev in events {
        if ev.get("ph").and_then(Value::as_str) != Some("C") {
            continue;
        }
        let Some(name) = ev.get("name").and_then(Value::as_str) else { continue };
        // perf_trace writes the running total under args.<name>; read the one numeric arg.
        if let Some(v) = ev.get("args").and_then(Value::as_object).and_then(|a| a.values().find_map(Value::as_u64)) {
            m.insert(name.to_string(), v);
        }
    }
    m
}

pub(crate) fn count(m: &BTreeMap<String, u64>, name: &str) -> u64 {
    m.get(name).copied().unwrap_or(0)
}

/// One row of spec §4.1.3 / §4.4.8, in the table's column order.
pub(crate) struct Row { pub program: u64, pub bcg: u64, pub cone: u64, pub v2: u64, pub scc: u64, pub pd: u64, pub ctx: u64, pub snap: u64 }

pub(crate) fn assert_row(case: &str, m: &BTreeMap<String, u64>, r: Row) {
    let got = [
        ("build.program", count(m, "build.program"), r.program),
        ("build.combined_graph", count(m, "build.combined_graph"), r.bcg),
        ("build.cone", count(m, "build.cone"), r.cone),
        ("build.summaries_v2", count(m, "build.summaries_v2"), r.v2),
        ("build.graph_scc", count(m, "build.graph_scc"), r.scc),
        ("build.pd_solve", count(m, "build.pd_solve"), r.pd),
        ("build.detector_context", count(m, "build.detector_context"), r.ctx),
        ("build.snapshot", count(m, "build.snapshot"), r.snap),
    ];
    for (name, have, want) in got {
        assert_eq!(have, want, "[{case}] {name}: got {have}, want {want}\nall counters: {m:?}");
    }
}
```
